const frames = [];
function clone(id,key,title,width=1366,height=768) {
  const node = pixso.getNodeById(id).clone();
  section.appendChild(node);
  node.name=key+' · '+title;
  node.x=36+(frames.length%3)*1530;
  node.y=56+Math.floor(frames.length/3)*950;
  if(node.width!==width || node.height!==height)node.resize(width,height);
  frames.push({id:node.id,key,title,width:node.width,height:node.height});
  return node;
}
function rewrite(node,value,metric) {
  if(!node)throw Error('Missing text '+value);
  const style = metric?.[2]??node.fontName.style??'Regular';
  if(node.fontName.family==='Golos Text'){
    node.textStyleId='';
    node.fontName=font(style);
  }
  if(metric){
    node.fontSize=metric[0];
    node.lineHeight={unit:'PIXELS',value:metric[1]};
  }
  node.characters=String(value);
  if(node.characters!==String(value)||node.hasMissingFont)throw Error('Invalid text '+node.id);
}
function panel(root){return root.findOne(n=>n.name==='Ritual · compact progress');}
function phase(root){return root.findOne(n=>n.name==='Current phase');}
function replaceRitual(root,reference) {
  const old=panel(root);
  const parent=old.parent;
  const index=parent.children.findIndex(n=>n.id===old.id);
  const copy=panel(pixso.getNodeById(reference)).clone();
  parent.insertChild(index,copy);
  old.remove();
  return copy;
}
function roundLabel(ritual,round) {
  const existing=ritual.children.find(n=>n.name==='Round indicator');
  if(existing){rewrite(existing,round+' / 7 РАУНДОВ',[13,18,'Regular']);return;}
  const badge=ritual.children.find(n=>n.name.startsWith('Tag /'));
  const copy=ritual.children.find(n=>n.name.startsWith('Small / Осталось')).clone();
  ritual.insertChild(ritual.children.findIndex(n=>n.id===badge.id)+1,copy);
  badge.visible=false;
  copy.name='Round indicator';
  rewrite(copy,round+' / 7 РАУНДОВ',[13,18,'Regular']);
}
function candleCounts(ritual,counts) {
  const route=ritual.children.find(n=>n.name==='Candle route');
  for(const [i,tile] of route.children.entries()){
    rewrite(tile.findOne(n=>n.type==='TEXT'&&n.name.startsWith('Code /')),counts[i]);
  }
  rewrite(ritual.children.find(n=>n.name.startsWith('Metric /')),3*counts[0]+2*counts[1]+counts[2]);
}
function scene(root,title,body) {
  const center=phase(root);
  if(title)rewrite(center.children.find(n=>n.type==='TEXT'&&n.name.startsWith('Title /')),title,[32,40,'Bold']);
  if(body)rewrite(center.children.find(n=>n.type==='TEXT'&&n.name.startsWith('Body /')),body,[16,24,'Regular']);
}
function named(root,prefix){return root.findOne(n=>n.type==='TEXT'&&n.name.startsWith(prefix));}

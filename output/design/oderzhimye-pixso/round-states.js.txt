function textNode(root, value) {
  const node=root.findOne(n=>n.type==='TEXT'&&n.characters===value);
  if(!node)throw Error('Missing text '+value);
  return node;
}
function write(node,value){
  if(node.fontName.family==='Golos Text')node.fontName={family:'Inter',style:node.fontName.style};
  node.characters=value;
  if(node.characters!==value)throw Error('Text mismatch '+node.id);
}
function phase(root){return root.findOne(n=>n.name==='Current phase');}
function ritual(root){return root.findOne(n=>n.name==='Ritual · compact progress');}
function mainBody(root,copy){
  const node=phase(root).children.find(n=>n.type==='TEXT'&&n.name.startsWith('Body'));
  if(!node)throw Error('Missing phase body');
  write(node,copy);
}
function snapshot(root,reference){
  const old=ritual(root);
  const parent=old.parent;
  const pos=parent.children.findIndex(n=>n.id===old.id);
  const replacement=ritual(pixso.getNodeById(R.nodes[reference])).clone();
  parent.insertChild(pos,replacement);
  old.remove();
  return replacement;
}
function label(root,old,newValue){write(textNode(root,old),newValue);}

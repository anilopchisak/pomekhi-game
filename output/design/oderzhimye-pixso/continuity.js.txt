function write(text,value){
  const previous=text.fontName;
  if(previous.family==='Golos Text'){
    text.textStyleId='';
    text.fontName=font;
    text.fontSize=16;
    text.lineHeight={unit:'PIXELS',value:24};
  }
  text.characters=String(value);
  if(text.characters!==String(value))throw Error('Text mismatch '+text.id);
}
function panel(root){return root.findOne(n=>n.name==='Ritual · compact progress');}
function setPanel(root,reference){
  const old=panel(root);
  const parent=old.parent;
  const index=parent.children.findIndex(c=>c.id===old.id);
  const copy=panel(pixso.getNodeById(reference)).clone();
  parent.insertChild(index,copy);
  old.remove();
  return copy;
}
function setRound(panel,round){
  const badge=panel.children.find(n=>n.name.startsWith('Tag /'));
  const roundText=panel.children.find(n=>n.name==='Round indicator')??panel.children.find(n=>n.name.startsWith('Small / Осталось')).clone();
  if(roundText.parent.id!==panel.id)panel.insertChild(panel.children.findIndex(n=>n.id===badge.id)+1,roundText);
  badge.visible=false;
  roundText.name='Round indicator';
  write(roundText,round+' / 7 РАУНДОВ');
  roundText.fontSize=13;
  roundText.lineHeight={unit:'PIXELS',value:18};
}
function setRitual(panel,counts){
  const candles=panel.children.find(n=>n.name==='Candle route').children;
  if(candles.length!==3)throw Error('Unexpected candle route');
  for(const [i,candle]of candles.entries()){
    const count=candle.findOne(n=>n.type==='TEXT'&&n.name.startsWith('Code /'));
    write(count,counts[i]);
    const flame=candle.findOne(n=>n.name==='Flame');
    const glow=candle.findOne(n=>n.name==='Local glow');
    if(flame)flame.opacity=counts[i]>0?1:0;
    if(glow)glow.opacity=counts[i]>0?.08:0;
  }
  write(panel.children.find(n=>n.name.startsWith('Metric /')),3*counts[0]+2*counts[1]+counts[2]);
}
function body(root,value){
  write(root.findOne(n=>n.name==='Current phase').children.find(n=>n.type==='TEXT'&&n.name.startsWith('Body')),value);
}
function resume(root,component){
  const timer=root.findOne(n=>n.type==='INSTANCE'&&n.name.startsWith('Phase timer'));
  timer.swapComponent(pixso.getNodeById(component));
  const clock=timer.findOne(n=>n.name==='Clock value');
  if(clock.characters!=='02:18')throw Error('Resume clock mismatch '+clock.characters);
  return clock.id;
}

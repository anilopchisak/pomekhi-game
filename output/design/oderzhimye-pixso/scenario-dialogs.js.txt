function paint(hex){
  const n=parseInt(hex,16);
  return {type:'SOLID',color:{r:(n>>16)/255,g:((n>>8)&255)/255,b:(n&255)/255}};
}

function stack(name,width,height,fill,pad,gap){
  const node=pixso.createFrame();
  node.name=name;
  node.resize(width,height);
  node.layoutMode='VERTICAL';
  node.primaryAxisSizingMode='FIXED';
  node.counterAxisSizingMode='FIXED';
  node.paddingLeft=node.paddingRight=node.paddingTop=node.paddingBottom=pad;
  node.itemSpacing=gap;
  node.fills=[palette[fill]];
  node.cornerRadius=12;
  node.strokes=[{...palette.border,opacity:.55}];
  return node;
}
function text(parent,content,size,style,color){
  const node=pixso.createText();
  node.name=content.slice(0,50);
  node.fontName=font(style);
  node.fontSize=size;
  node.lineHeight={unit:'PERCENT',value:135};
  node.fills=[palette[color]];
  node.characters=content;
  node.textAutoResize='HEIGHT';
  parent.appendChild(node);
  node.resize(parent.width-parent.paddingLeft-parent.paddingRight,node.height);
  if(node.characters!==content || node.hasMissingFont)throw Error('Invalid text '+node.id);
  return node;
}

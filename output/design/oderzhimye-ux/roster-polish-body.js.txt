const p=await figma.getNodeByIdAsync(D.pages.screens);await figma.setCurrentPageAsync(p);const mutated=[],hoverIssues=[];
function reflow(n,w){if(n.type==='TEXT'){n.resize(w,n.height);n.textAutoResize='HEIGHT';mutated.push(n.id);return;}if(n.type==='INSTANCE'){if(n.name.startsWith('Tag / '))return;n.resize(w,n.height);mutated.push(n.id);return;}if(n.type!=='FRAME')return;const oldW=n.width,oldInner=oldW-n.paddingLeft-n.paddingRight,v=n.layoutSizingVertical;n.resize(w,n.height);if(v==='FILL'||v==='HUG')n.layoutSizingVertical=v;const iw=w-n.paddingLeft-n.paddingRight;
 if(n.name==='Ritual signal'){for(const c of n.children)if(c.type==='ELLIPSE'&&c.layoutPositioning==='ABSOLUTE')c.x=(w-c.width)/2;return;}
 if(n.name==='Ritual positions'){const oldSW=(oldW-24)/3,newSW=(w-24)/3;for(const c of n.children){if(c.name.startsWith('Prophecy '))reflow(c,newSW);else if(c.name.startsWith('Candle ')){const stage=Math.min(2,Math.floor(c.x/(oldSW+12)));const col=Math.round((c.x-stage*(oldSW+12)-14)/21);c.x=stage*(newSW+12)+14+col*20;}else if(c.name==='Final candle smoke')c.x=2*(newSW+12)+18;}return;}
 const normal=n.children.filter(c=>c.layoutPositioning!=='ABSOLUTE');if(n.layoutMode==='VERTICAL')for(const c of normal)reflow(c,iw);else if(n.layoutMode==='HORIZONTAL'&&normal.length){const width=(iw-(normal.length-1)*n.itemSpacing)/normal.length;for(const c of normal)reflow(c,width);}mutated.push(n.id);}
for(const[id,nid]of Object.entries(R.nodes)){if(!id.startsWith('L')&&!/^[AB]/.test(id))continue;const f=await figma.getNodeByIdAsync(nid);const roster=f?.findOne(n=>n.name==='Participants · stable original roles'),center=f?.findOne(n=>n.name==='Current phase');if(!roster||!center)continue;roster.resize(340,roster.height);const head=roster.children.find(n=>n.name==='Participants header');if(head){head.resize(308,head.height);const title=head.children.find(n=>n.type==='TEXT');title.resize(226,title.height);}for(const row of roster.children){if(row.type==='INSTANCE'&&row.name.startsWith('Participant / ')){row.resize(308,44);continue;}if(row.name.startsWith('Participant slot / ')){row.resize(308,44);const control=row.children.find(n=>n.type==='INSTANCE'&&n.name.startsWith('Action / Заменить'));const person=row.children.find(n=>n.type==='INSTANCE'&&n.name.startsWith('Participant / '));if(person)person.resize(control?264:308,44);if(control)control.x=264;}if(row.name==='Privacy footer'){row.resize(308,row.height);for(const t of row.children)if(t.type==='TEXT')t.resize(308,t.height);}}
 reflow(center,center.width-40);mutated.push(roster.id,center.id);}
// Explicit hover reactions retain the click action on the root instance.
for(const[id,a]of Object.entries(R.actions))for(const key of Object.keys(a).filter(k=>/^replace\d+$/.test(k))){const n=await figma.getNodeByIdAsync(a[key]);if(!n)continue;const click=n.reactions.filter(r=>r.trigger?.type==='ON_CLICK');try{await n.setReactionsAsync([...click,{trigger:{type:'ON_HOVER'},actions:[{type:'NODE',destinationId:RC.Hover,navigation:'CHANGE_TO',transition:{type:'DISSOLVE',duration:.18,easing:{type:'EASE_OUT'}}}]},{trigger:{type:'MOUSE_LEAVE',delay:0},actions:[{type:'NODE',destinationId:RC.Default,navigation:'CHANGE_TO',transition:{type:'DISSOLVE',duration:.18,easing:{type:'EASE_OUT'}}}]}]);mutated.push(n.id);}catch(e){hoverIssues.push({id,key,error:String(e)});break;}}
return {mutatedNodeIds:mutated,hoverIssues};

ObjC.import('CoreGraphics');
ObjC.import('AppKit');
function run(argv) {
 const code=$.NSString.stringWithContentsOfFileEncodingError(argv[0],$.NSUTF8StringEncoding,null).js;
 const pb=$.NSPasteboard.generalPasteboard;pb.clearContents;pb.setStringForType(code,$.NSPasteboardTypeString);
 const safari=Application('Safari');safari.activate();safari.windows[0].bounds={x:218,y:58,width:1306,height:871};delay(0.7);
 const sys=Application('System Events');
 function click(x,y){const p=$.CGPointMake(x,y);$.CGEventPost(0,$.CGEventCreateMouseEvent(null,1,p,0));$.CGEventPost(0,$.CGEventCreateMouseEvent(null,2,p,0));delay(0.3);}
 click(800,400);sys.keyCode(0,{using:['command down']});sys.keyCode(9,{using:['command down']});delay(0.8);
 sys.keyCode(0,{using:['command down']});sys.keyCode(8,{using:['command down']});delay(0.4);
 const pasted=pb.stringForType($.NSPasteboardTypeString).js;if(pasted.replace(/\r\n?/g,'\n').trim()!==code.replace(/\r\n?/g,'\n').trim())return JSON.stringify({error:'Paste verification failed',expectedLength:code.length,actualLength:pasted.length});
 click(590,600);delay(Number(argv[1]||2));
 click(800,700);sys.keyCode(0,{using:['command down']});sys.keyCode(8,{using:['command down']});delay(0.2);
 return pb.stringForType($.NSPasteboardTypeString).js;
}

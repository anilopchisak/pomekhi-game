figma.ui.onmessage=async message=>{
 if(message.type!=='pomekhi-run')return;
 try{
  const AsyncFunction=Object.getPrototypeOf(async function(){}).constructor;
  const result=await new AsyncFunction('figma',message.code)(figma);
  figma.ui.postMessage({type:'pomekhi-result',ok:true,result});
 }catch(error){figma.ui.postMessage({type:'pomekhi-result',ok:false,result:{error:String(error),stack:error.stack}});}
};
